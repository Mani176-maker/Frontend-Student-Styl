async function pay() { const name = document.getElementById("name").value; const phone = document.getElementById("phone").value; const files = document.getElementById("files").files; const amount = document.getElementById("amount").value; if (!name || !phone || files.length === 0) { alert("Please fill all details and upload a file."); return; } const formData = new FormData(); formData.append("files", files[0]); formData.append("name", name); formData.append("phone", phone); formData.append("amount", amount); try { const response = await fetch(`${BASE_URL}/order`, { method: "POST", body: formData }); const d = await response.json(); const rzp = new Razorpay({ key: d.key, amount: d.amount * 100, currency: "INR", order_id: d.orderId, handler: function(res) { const driveLink = `https://drive.google.com/file/d/${d.driveFileId}/view`; const message = `*NEW PRINT ORDER* ✅%0A%0A` + `*Customer:* ${name}%0A` + `*Phone:* ${phone}%0A` + `*Total:* ₹${amount}%0A%0A` + `*📂 PDF LINK:* ${driveLink}%0A%0A` + `*Payment ID:* ${res.razorpay_payment_id}`; window.open(`https://wa.me/917569008169?text=${message}`, '_blank'); } }); rzp.open(); } catch (err) { alert("Payment failed to initialize."); } }
