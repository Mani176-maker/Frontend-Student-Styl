async function pay() { const name = document.getElementById("name").value; const phone = document.getElementById("phone").value; const files = document.getElementById("files").files; const amount = document.getElementById("amount").value; const printType = document.getElementById("printType").value; const bindingType = document.getElementById("bindingType").value; if (!name || !phone || files.length === 0) { alert("Please fill all details and upload a PDF."); return; } const payBtn = document.getElementById("payBtn"); const originalText = payBtn.innerText; // Disable button and show status payBtn.innerText = "Uploading PDF..."; payBtn.disabled = true; const formData = new FormData(); formData.append("files", files[0]); formData.append("name", name); formData.append("phone", phone); formData.append("amount", amount); formData.append("printType", printType); try { const response = await fetch(`${BASE_URL}/order`, { method: "POST", body: formData }); if (!response.ok) throw new Error("Upload failed. Check server logs."); const d = await response.json(); const rzp = new Razorpay({ key: d.key, amount: d.amount * 100, currency: "INR", order_id: d.orderId, handler: function(res) { // Construct the Link and Message for WhatsApp const driveLink = `https://drive.google.com/file/d/${d.driveFileId}/view`; const message = `*NEW PRINT ORDER* ✅%0A%0A` + `*Customer:* ${name}%0A` + `*Phone:* ${phone}%0A` + `*Total Paid:* ₹${amount}%0A` + `*Print Type:* ${printType}%0A%0A` + `*📂 PDF LINK:* ${driveLink}%0A%0A` + `*Payment ID:* ${res.razorpay_payment_id}`; // Open WhatsApp to your business number window.open(`https://wa.me/917569008169?text=${message}`, '_blank'); payBtn.innerText = originalText; payBtn.disabled = false; }, "modal": { "ondismiss": function(){ payBtn.innerText = originalText; payBtn.disabled = false; } } }); rzp.open(); } catch (err) { alert("Error: " + err.message); payBtn.innerText = originalText; payBtn.disabled = false; } }
